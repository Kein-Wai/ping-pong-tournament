// src/api/axios.ts
import axios from "axios";
import AsyncStorage from "@react-native-async-storage/async-storage";

// Expo inyecta automáticamente las variables que empiezan por EXPO_PUBLIC_
const API_URL = process.env.EXPO_PUBLIC_API_URL;

export const api = axios.create({
  baseURL: API_URL,
  headers: {
    "Content-Type": "application/json",
  },
});

// Interceptor para inyectar el token en cada petición (si existe)
api.interceptors.request.use(
  async (config) => {
    const token = await AsyncStorage.getItem("pingpong_token");
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error),
);

// src/navigation/MainTabNavigator.tsx
import React from "react";
import { createBottomTabNavigator } from "@react-navigation/bottom-tabs";
import { Ionicons } from "@expo/vector-icons"; // Viene gratis con Expo

// Importamos las pantallas
import DashboardScreen from "../screens/DashboardScreen";
import TrainingsScreen from "../screens/TrainingsScreen";
import TrackerScreen from "../screens/TrackerScreen";
import GamesScreen from "../screens/GamesScreen";

const Tab = createBottomTabNavigator();

export default function MainTabNavigator() {
  return (
    <Tab.Navigator
      screenOptions={({ route }) => ({
        headerShown: false, // Ocultamos la cabecera fea por defecto
        tabBarStyle: {
          backgroundColor: "#1f2937",
          borderTopColor: "#374151",
          paddingBottom: 5,
          height: 60,
        },
        tabBarActiveTintColor: "#3b82f6",
        tabBarInactiveTintColor: "#9ca3af",
        tabBarIcon: ({ focused, color, size }) => {
          let iconName: keyof typeof Ionicons.glyphMap = "help-circle";

          if (route.name === "Inicio") {
            iconName = focused ? "home" : "home-outline";
          } else if (route.name === "Entrenos") {
            iconName = focused ? "barbell" : "barbell-outline";
          } else if (route.name === "Tracker") {
            iconName = focused ? "analytics" : "analytics-outline";
          } else if (route.name === "Juegos") {
            iconName = focused ? "game-controller" : "game-controller-outline";
          }

          return <Ionicons name={iconName} size={size} color={color} />;
        },
      })}
    >
      <Tab.Screen name="Inicio" component={DashboardScreen} />
      <Tab.Screen name="Entrenos" component={TrainingsScreen} />
      <Tab.Screen name="Tracker" component={TrackerScreen} />
      <Tab.Screen name="Juegos" component={GamesScreen} />
    </Tab.Navigator>
  );
}

// src/navigation/RootNavigator.tsx
import React from "react";
import { createNativeStackNavigator } from "@react-navigation/native-stack";
import { useAuthStore } from "../store/authStore";

import LoginScreen from "../screens/LoginScreen";
import MainTabNavigator from "./MainTabNavigator";

const Stack = createNativeStackNavigator();

export default function RootNavigator() {
  const { isAuthenticated } = useAuthStore();

  return (
    <Stack.Navigator screenOptions={{ headerShown: false }}>
      {isAuthenticated ? (
        // Si está logueado, le enseñamos las pestañas
        <Stack.Screen name="MainApp" component={MainTabNavigator} />
      ) : (
        // Si NO está logueado, a la pantalla de Login
        <Stack.Screen name="Login" component={LoginScreen} />
      )}
    </Stack.Navigator>
  );
}

import React, { useState, useEffect, useMemo } from "react";
import {
  View,
  Text,
  StyleSheet,
  ScrollView,
  TouchableOpacity,
  TextInput,
  Modal,
  ActivityIndicator,
  RefreshControl,
  Dimensions,
} from "react-native";
import { SafeAreaView } from "react-native-safe-area-context";
import { Ionicons } from "@expo/vector-icons";
import { api } from "../api/axios";

const { width } = Dimensions.get("window");
const ITEMS_PER_PAGE = 8;

interface ManualMatch {
  id: string;
  opponentName: string;
  matchType: string;
  location: string;
  format: string;
  analysisType: "Rapido" | "Extenso";
  opponentHand?: string | null;
  opponentStyle?: string | null;
  opponentLevel?: string | null;
  opponentRubberForehand?: string | null;
  opponentRubberBackhand?: string | null;
  status: "Programado" | "Iniciado" | "Completado" | "Cancelado";
  setsToWin: number;
  mySets: number;
  opponentSets: number;
  date: string;
}

interface Props {
  navigation?: any;
  onSelectMatch?: (matchId: string) => void;
}

export default function AnalisisListScreen({
  navigation,
  onSelectMatch,
}: Props) {
  const [matches, setMatches] = useState<ManualMatch[]>([]);
  const [loading, setLoading] = useState(true);
  const [refreshing, setRefreshing] = useState(false);
  const [currentSeason, setCurrentSeason] = useState<string>("");

  // Filtros y tabs
  const [activeTab, setActiveTab] = useState<"analisis" | "planes">("analisis");
  const [search, setSearch] = useState("");
  const [page, setPage] = useState(1);

  // Estados del Modal de Creación
  const [modalOpen, setModalOpen] = useState(false);
  const [creating, setCreating] = useState(false);
  const [initialStatus, setInitialStatus] = useState<"Programado" | "Iniciado">(
    "Programado",
  );
  const [opponentName, setOppName] = useState("");
  const [matchType, setMatchType] = useState("Amistoso");
  const [matchFormat, setMatchFormat] = useState("Individual");
  const [analysisType, setAnalysisType] = useState<"Rapido" | "Extenso">(
    "Rapido",
  );
  const [setsToWin, setSetsToWin] = useState("3");
  const [opponentLevel, setOpponentLevel] = useState<string | null>(null);
  const [opponentHand, setOpponentHand] = useState<string | null>(null);
  const [opponentStyle, setOpponentStyle] = useState<string | null>(null);
  const [opponentRubberForehand, setOpponentRubberForehand] = useState<
    string | null
  >(null);
  const [opponentRubberBackhand, setOpponentRubberBackhand] = useState<
    string | null
  >(null);

  const fetchMatchesAndSeason = async () => {
    try {
      const [matchesRes, seasonsRes] = await Promise.all([
        api.get("/manual-matches"),
        api.get("/seasons"),
      ]);

      const matchData = matchesRes.data.data || matchesRes.data || [];
      setMatches(matchData);

      const seasons = seasonsRes.data.data || seasonsRes.data || [];
      const activeS = seasons.find((s: any) => s.isCurrent);
      if (activeS) setCurrentSeason(activeS.name);
    } catch (error) {
      console.error("Error cargando partidos y temporada:", error);
    } finally {
      setLoading(false);
      setRefreshing(false);
    }
  };

  useEffect(() => {
    fetchMatchesAndSeason();
  }, []);

  const onRefresh = () => {
    setRefreshing(true);
    fetchMatchesAndSeason();
  };

  const handleCreateMatch = async () => {
    setCreating(true);
    const finalOpponent =
      opponentName.trim() === "" ? "Rival por determinar" : opponentName.trim();

    try {
      const payload = {
        opponentName: finalOpponent,
        matchType,
        location: "Casa",
        format: matchFormat,
        analysisType: initialStatus === "Programado" ? "Rapido" : analysisType,
        opponentHand,
        opponentStyle,
        opponentLevel,
        opponentRubberForehand,
        opponentRubberBackhand,
        status: initialStatus,
        setsToWin: Number(setsToWin),
        date: new Date().toISOString(),
      };

      const res = await api.post("/manual-matches", payload);
      const newMatch = res.data.data || res.data;
      setModalOpen(false);
      resetForm();
      await fetchMatchesAndSeason();

      // Navegar al Tracker
      if (onSelectMatch) {
        onSelectMatch(newMatch.id);
      } else if (navigation?.navigate) {
        navigation.navigate("ManualMatchTracker", { id: newMatch.id });
      }
    } catch (error) {
      console.error("Error creando partido:", error);
    } finally {
      setCreating(false);
    }
  };

  const resetForm = () => {
    setOppName("");
    setInitialStatus("Programado");
    setMatchType("Amistoso");
    setMatchFormat("Individual");
    setAnalysisType("Rapido");
    setSetsToWin("3");
    setOpponentLevel(null);
    setOpponentHand(null);
    setOpponentStyle(null);
    setOpponentRubberForehand(null);
    setOpponentRubberBackhand(null);
  };

  const filteredMatches = useMemo(() => {
    return matches.filter((m) => {
      const matchSearch = m.opponentName
        ?.toLowerCase()
        .includes(search.toLowerCase());
      const matchTab =
        activeTab === "planes"
          ? m.status === "Programado"
          : m.status !== "Programado";
      return matchSearch && matchTab;
    });
  }, [matches, search, activeTab]);

  const totalPages = Math.ceil(filteredMatches.length / ITEMS_PER_PAGE);
  const paginatedMatches = useMemo(() => {
    return filteredMatches.slice(
      (page - 1) * ITEMS_PER_PAGE,
      page * ITEMS_PER_PAGE,
    );
  }, [filteredMatches, page]);

  const handleCardPress = (matchId: string) => {
    if (onSelectMatch) {
      onSelectMatch(matchId);
    } else if (navigation?.navigate) {
      navigation.navigate("ManualMatchTracker", { id: matchId });
    }
  };

  return (
    <SafeAreaView style={styles.container}>
      {/* 1. CABECERA */}
      <View style={styles.header}>
        <View style={styles.headerLeft}>
          <View style={styles.headerIconContainer}>
            <Ionicons name="analytics" size={24} color="#3b82f6" />
          </View>
          <View>
            <Text style={styles.headerTitle}>Análisis y Planes</Text>
            {currentSeason ? (
              <View style={styles.seasonBadge}>
                <Text style={styles.seasonBadgeText}>{currentSeason}</Text>
              </View>
            ) : null}
          </View>
        </View>

        <TouchableOpacity
          style={styles.newRegisterBtn}
          activeOpacity={0.8}
          onPress={() => setModalOpen(true)}
        >
          <Ionicons name="add" size={18} color="#fff" />
          <Text style={styles.newRegisterBtnText}>Nuevo</Text>
        </TouchableOpacity>
      </View>

      {/* 2. PESTAÑAS (Análisis y Resultados vs Planificaciones Futuras) */}
      <View style={styles.tabsRow}>
        <TouchableOpacity
          style={[
            styles.tabButton,
            activeTab === "analisis" && styles.activeTabButton,
          ]}
          onPress={() => {
            setActiveTab("analisis");
            setPage(1);
          }}
        >
          <Ionicons
            name="tennisball-outline"
            size={16}
            color={activeTab === "analisis" ? "#3b82f6" : "#9ca3af"}
          />
          <Text
            style={[
              styles.tabButtonText,
              activeTab === "analisis" && styles.activeTabButtonText,
            ]}
          >
            Análisis y Resultados
          </Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={[
            styles.tabButton,
            activeTab === "planes" && styles.activeTabButtonOrange,
          ]}
          onPress={() => {
            setActiveTab("planes");
            setPage(1);
          }}
        >
          <Ionicons
            name="clipboard-outline"
            size={16}
            color={activeTab === "planes" ? "#f59e0b" : "#9ca3af"}
          />
          <Text
            style={[
              styles.tabButtonText,
              activeTab === "planes" && styles.activeTabButtonTextOrange,
            ]}
          >
            Planificaciones Futuras
          </Text>
        </TouchableOpacity>
      </View>

      {/* 3. BARRA DE BÚSQUEDA */}
      <View style={styles.searchContainer}>
        <Ionicons
          name="search"
          size={18}
          color="#9ca3af"
          style={styles.searchIcon}
        />
        <TextInput
          placeholder="Buscar rival..."
          placeholderTextColor="#6b7280"
          value={search}
          onChangeText={(val) => {
            setSearch(val);
            setPage(1);
          }}
          style={styles.searchInput}
        />
        {search ? (
          <TouchableOpacity onPress={() => setSearch("")}>
            <Ionicons name="close-circle" size={18} color="#9ca3af" />
          </TouchableOpacity>
        ) : null}
      </View>

      {/* 4. LISTA DE PARTIDOS PAGINADA */}
      {loading ? (
        <View style={styles.loaderContainer}>
          <ActivityIndicator size="large" color="#3b82f6" />
        </View>
      ) : (
        <ScrollView
          contentContainerStyle={styles.scrollList}
          showsVerticalScrollIndicator={false}
          refreshControl={
            <RefreshControl
              refreshing={refreshing}
              onRefresh={onRefresh}
              tintColor="#3b82f6"
            />
          }
        >
          {paginatedMatches.length === 0 ? (
            <View style={styles.emptyContainer}>
              <Ionicons
                name={
                  activeTab === "planes"
                    ? "clipboard-outline"
                    : "tennisball-outline"
                }
                size={44}
                color="#4b5563"
              />
              <Text style={styles.emptyText}>
                No hay {activeTab === "planes" ? "planificaciones" : "partidos"}{" "}
                registrados.
              </Text>
            </View>
          ) : (
            paginatedMatches.map((m) => {
              const isPlan = m.status === "Programado";
              const isCompleted = m.status === "Completado";

              const statusBg = isPlan
                ? "rgba(245, 158, 11, 0.15)"
                : isCompleted
                  ? "rgba(16, 185, 129, 0.15)"
                  : "rgba(59, 130, 246, 0.15)";
              const statusColor = isPlan
                ? "#f59e0b"
                : isCompleted
                  ? "#10b981"
                  : "#3b82f6";

              const buttonLabel = isCompleted
                ? "Ver Reporte"
                : isPlan
                  ? "Planificar / Jugar"
                  : "Continuar Arbitraje";

              return (
                <View
                  key={m.id}
                  style={[
                    styles.matchCard,
                    isPlan ? styles.cardBorderOrange : styles.cardBorderBlue,
                  ]}
                >
                  <View style={styles.cardHeader}>
                    <View style={styles.cardHeaderTitleRow}>
                      <Text style={styles.opponentTitle}>
                        vs {m.opponentName}
                      </Text>
                      <View
                        style={[
                          styles.statusBadge,
                          { backgroundColor: statusBg },
                        ]}
                      >
                        <Text
                          style={[
                            styles.statusBadgeText,
                            { color: statusColor },
                          ]}
                        >
                          {m.status}
                        </Text>
                      </View>
                    </View>
                    <Text style={styles.cardSubDetails}>
                      {m.date
                        ? new Date(m.date).toLocaleDateString("es-ES")
                        : "Sin fecha"}{" "}
                      · Formato: {m.format} · {m.matchType} ({m.analysisType})
                    </Text>
                  </View>

                  <View style={styles.cardActionRow}>
                    <TouchableOpacity
                      style={[
                        styles.cardActionButton,
                        isPlan ? styles.btnOrangeLight : styles.btnBlueLight,
                      ]}
                      onPress={() => handleCardPress(m.id)}
                      activeOpacity={0.8}
                    >
                      <Text
                        style={[
                          styles.cardActionButtonText,
                          { color: isPlan ? "#f59e0b" : "#3b82f6" },
                        ]}
                      >
                        {buttonLabel}
                      </Text>
                      <Ionicons
                        name="chevron-forward"
                        size={16}
                        color={isPlan ? "#f59e0b" : "#3b82f6"}
                      />
                    </TouchableOpacity>
                  </View>
                </View>
              );
            })
          )}

          {/* Paginación */}
          {totalPages > 1 && (
            <View style={styles.paginationRow}>
              <TouchableOpacity
                disabled={page <= 1}
                onPress={() => setPage((prev) => Math.max(prev - 1, 1))}
                style={[styles.pageBtn, page <= 1 && styles.pageBtnDisabled]}
              >
                <Ionicons
                  name="chevron-back"
                  size={16}
                  color={page <= 1 ? "#4b5563" : "#fff"}
                />
              </TouchableOpacity>
              <Text style={styles.pageIndicator}>
                Página {page} de {totalPages}
              </Text>
              <TouchableOpacity
                disabled={page >= totalPages}
                onPress={() =>
                  setPage((prev) => Math.min(prev + 1, totalPages))
                }
                style={[
                  styles.pageBtn,
                  page >= totalPages && styles.pageBtnDisabled,
                ]}
              >
                <Ionicons
                  name="chevron-forward"
                  size={16}
                  color={page >= totalPages ? "#4b5563" : "#fff"}
                />
              </TouchableOpacity>
            </View>
          )}
        </ScrollView>
      )}

      {/* 5. MODAL: CONFIGURAR PARTIDO / PLAN */}
      <Modal visible={modalOpen} animationType="slide" transparent={true}>
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <View style={styles.modalTopBar}>
              <Text style={styles.modalTitle}>Configurar Partido / Plan</Text>
              <TouchableOpacity onPress={() => setModalOpen(false)}>
                <Ionicons name="close" size={24} color="#9ca3af" />
              </TouchableOpacity>
            </View>

            <ScrollView
              showsVerticalScrollIndicator={false}
              contentContainerStyle={styles.modalScroll}
            >
              {/* Segmented Control: Modo */}
              <View style={styles.segmentedControl}>
                <TouchableOpacity
                  style={[
                    styles.segmentButton,
                    initialStatus === "Programado" &&
                      styles.segmentButtonActiveOrange,
                  ]}
                  onPress={() => setInitialStatus("Programado")}
                >
                  <Text
                    style={[
                      styles.segmentText,
                      initialStatus === "Programado" &&
                        styles.segmentTextActive,
                    ]}
                  >
                    Planificar Pre-Partido
                  </Text>
                </TouchableOpacity>

                <TouchableOpacity
                  style={[
                    styles.segmentButton,
                    initialStatus === "Iniciado" &&
                      styles.segmentButtonActiveBlue,
                  ]}
                  onPress={() => setInitialStatus("Iniciado")}
                >
                  <Text
                    style={[
                      styles.segmentText,
                      initialStatus === "Iniciado" && styles.segmentTextActive,
                    ]}
                  >
                    Empezar a Jugar Ya
                  </Text>
                </TouchableOpacity>
              </View>

              {/* Si es iniciado: Nivel de Detalle */}
              {initialStatus === "Iniciado" && (
                <View style={styles.inputGroup}>
                  <Text style={styles.inputLabel}>Nivel de Detalle</Text>
                  <View style={styles.pillRow}>
                    <TouchableOpacity
                      style={[
                        styles.pillOption,
                        analysisType === "Rapido" && styles.pillActive,
                      ]}
                      onPress={() => setAnalysisType("Rapido")}
                    >
                      <Text
                        style={[
                          styles.pillText,
                          analysisType === "Rapido" && styles.pillTextActive,
                        ]}
                      >
                        Rápido (Marcador global)
                      </Text>
                    </TouchableOpacity>
                    <TouchableOpacity
                      style={[
                        styles.pillOption,
                        analysisType === "Extenso" && styles.pillActive,
                      ]}
                      onPress={() => setAnalysisType("Extenso")}
                    >
                      <Text
                        style={[
                          styles.pillText,
                          analysisType === "Extenso" && styles.pillTextActive,
                        ]}
                      >
                        Extenso (Punto a Punto)
                      </Text>
                    </TouchableOpacity>
                  </View>
                </View>
              )}

              {/* Nombre del Rival */}
              <View style={styles.inputGroup}>
                <Text style={styles.inputLabel}>
                  Nombre del Rival {initialStatus === "Iniciado" && "*"}
                </Text>
                {initialStatus === "Programado" && (
                  <Text style={styles.inputDesc}>
                    Déjalo en blanco si aún no sabes contra quién juegas
                  </Text>
                )}
                <TextInput
                  placeholder="Ej. Juan Pérez"
                  placeholderTextColor="#6b7280"
                  value={opponentName}
                  onChangeText={setOppName}
                  style={styles.textInput}
                />
              </View>

              {/* Formato y Tipo de Partido */}
              <View style={styles.rowTwoCols}>
                <View style={styles.colHalf}>
                  <Text style={styles.inputLabel}>Tipo de Partido</Text>
                  <View style={styles.pillCol}>
                    {["Amistoso", "Liga", "Competicion"].map((t) => (
                      <TouchableOpacity
                        key={t}
                        style={[
                          styles.miniPill,
                          matchType === t && styles.pillActive,
                        ]}
                        onPress={() => setMatchType(t)}
                      >
                        <Text
                          style={[
                            styles.pillText,
                            matchType === t && styles.pillTextActive,
                          ]}
                        >
                          {t}
                        </Text>
                      </TouchableOpacity>
                    ))}
                  </View>
                </View>

                <View style={styles.colHalf}>
                  <Text style={styles.inputLabel}>Formato</Text>
                  <View style={styles.pillCol}>
                    {["Individual", "Equipos"].map((f) => (
                      <TouchableOpacity
                        key={f}
                        style={[
                          styles.miniPill,
                          matchFormat === f && styles.pillActive,
                        ]}
                        onPress={() => setMatchFormat(f)}
                      >
                        <Text
                          style={[
                            styles.pillText,
                            matchFormat === f && styles.pillTextActive,
                          ]}
                        >
                          {f}
                        </Text>
                      </TouchableOpacity>
                    ))}
                  </View>
                </View>
              </View>

              {/* Sets para ganar */}
              <View style={styles.inputGroup}>
                <Text style={styles.inputLabel}>Formato al mejor de...</Text>
                <View style={styles.pillRow}>
                  {[
                    { val: "1", label: "1 Set" },
                    { val: "2", label: "Mejor 3 (Gana 2)" },
                    { val: "3", label: "Mejor 5 (Gana 3)" },
                    { val: "4", label: "Mejor 7 (Gana 4)" },
                  ].map((s) => (
                    <TouchableOpacity
                      key={s.val}
                      style={[
                        styles.pillOptionSmall,
                        setsToWin === s.val && styles.pillActive,
                      ]}
                      onPress={() => setSetsToWin(s.val)}
                    >
                      <Text
                        style={[
                          styles.pillText,
                          setsToWin === s.val && styles.pillTextActive,
                        ]}
                      >
                        {s.label}
                      </Text>
                    </TouchableOpacity>
                  ))}
                </View>
              </View>

              {/* Atributos del Rival: Mano y Estilo */}
              <View style={styles.rowTwoCols}>
                <View style={styles.colHalf}>
                  <Text style={styles.inputLabel}>Mano del Rival</Text>
                  <View style={styles.pillRow}>
                    {["Diestro", "Zurdo"].map((h) => (
                      <TouchableOpacity
                        key={h}
                        style={[
                          styles.miniPill,
                          opponentHand === h && styles.pillActive,
                        ]}
                        onPress={() =>
                          setOpponentHand(opponentHand === h ? null : h)
                        }
                      >
                        <Text
                          style={[
                            styles.pillText,
                            opponentHand === h && styles.pillTextActive,
                          ]}
                        >
                          {h}
                        </Text>
                      </TouchableOpacity>
                    ))}
                  </View>
                </View>

                <View style={styles.colHalf}>
                  <Text style={styles.inputLabel}>Estilo del Rival</Text>
                  <View style={styles.pillRow}>
                    {["Ofensivo", "Defensivo"].map((st) => (
                      <TouchableOpacity
                        key={st}
                        style={[
                          styles.miniPill,
                          opponentStyle === st && styles.pillActive,
                        ]}
                        onPress={() =>
                          setOpponentStyle(opponentStyle === st ? null : st)
                        }
                      >
                        <Text
                          style={[
                            styles.pillText,
                            opponentStyle === st && styles.pillTextActive,
                          ]}
                        >
                          {st}
                        </Text>
                      </TouchableOpacity>
                    ))}
                  </View>
                </View>
              </View>

              {/* Nivel del Rival */}
              <View style={styles.inputGroup}>
                <Text style={styles.inputLabel}>Nivel del Rival</Text>
                <View style={styles.pillRow}>
                  {["Peor", "Igual", "Mejor"].map((lvl) => (
                    <TouchableOpacity
                      key={lvl}
                      style={[
                        styles.miniPill,
                        opponentLevel === lvl && styles.pillActive,
                      ]}
                      onPress={() =>
                        setOpponentLevel(opponentLevel === lvl ? null : lvl)
                      }
                    >
                      <Text
                        style={[
                          styles.pillText,
                          opponentLevel === lvl && styles.pillTextActive,
                        ]}

const formatRubberName = (rubberType?: string | null) => {
  if (!rubberType) return null;
  const map: Record<string, string> = {
    LISA: "Lisa",
    PICOS_CORTOS: "Picos Cortos",
    PICOS_MEDIOS: "Picos Medios",
    PICOS_LARGOS: "Picos Largos",
    ANTISPIN: "Antispin",
  };
  return map[rubberType] || rubberType;
};

// src/screens/LoginScreen.tsx
import React, { useState } from "react";
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  StyleSheet,
  ActivityIndicator,
  Alert,
} from "react-native";
import { api } from "../api/axios";
import { useAuthStore } from "../store/authStore";

export default function LoginScreen() {
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [loading, setLoading] = useState(false);
  const login = useAuthStore((state) => state.login);

  const handleLogin = async () => {
    if (!email || !password) {
      Alert.alert("Aviso", "Rellena todos los campos");
      return;
    }

    setLoading(true);
    try {
      // Usamos las rutas que limpiamos en el backend
      const response = await api.post("/auth/login", {
        email: email.toLowerCase().trim(),
        password,
      });

      await login(response.data.token);
    } catch (error: any) {
      const msg =
        error.response?.data?.error || "Error al conectar con el servidor";
      Alert.alert("Error de acceso", msg);
    } finally {
      setLoading(false);
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>TM Evolve</Text>
      <Text style={styles.subtitle}>Performance Tracker</Text>

      <View style={styles.form}>
        <TextInput
          style={styles.input}
          placeholder="Correo electrónico"
          placeholderTextColor="#6b7280"
          autoCapitalize="none"
          keyboardType="email-address"
          value={email}
          onChangeText={setEmail}
        />

        <TextInput
          style={styles.input}
          placeholder="Contraseña"
          placeholderTextColor="#6b7280"
          secureTextEntry
          value={password}
          onChangeText={setPassword}
        />

        <TouchableOpacity
          style={styles.button}
          onPress={handleLogin}
          disabled={loading}
        >
          {loading ? (
            <ActivityIndicator color="#ffffff" />
          ) : (
            <Text style={styles.buttonText}>Iniciar Sesión</Text>
          )}
        </TouchableOpacity>
      </View>
    </View>
  );
}

// src/store/authStore.ts
import { create } from "zustand";
import { api } from "../api/axios";
import AsyncStorage from "@react-native-async-storage/async-storage";

export interface User {
  id: string;
  email: string;
  name: string;
  surname?: string;
  role: string;
  clubId?: string | null;
  dominantHand?: string | null;
  playstyle?: string | null;
  rubberForehand?: string | null;
  rubberBackhand?: string | null;
  avatarUrl?: string | null;
  level?: string | null;

  // Añadimos Stats y Skills como opcionales
  stats?: any[] | any;
  skills?: any[] | any;
}

interface AuthState {
  token: string | null;
  user: User | null;
  isAuthenticated: boolean;
  isLoading: boolean;
  login: (token: string) => Promise<void>;
  logout: () => Promise<void>;
  fetchProfile: () => Promise<void>;
  initAuth: () => Promise<void>;
}

export const useAuthStore = create<AuthState>((set, get) => ({
  token: null,
  user: null,
  isAuthenticated: false,
  isLoading: true,

  login: async (token: string) => {
    await AsyncStorage.setItem("pingpong_token", token);
    set({ token, isAuthenticated: true });
    await get().fetchProfile();
  },

  logout: async () => {
    await AsyncStorage.removeItem("pingpong_token");
    set({ token: null, user: null, isAuthenticated: false });
  },

  fetchProfile: async () => {
    try {
      const response = await api.get("/users/me");
      const fullUser = response.data.data;

      set({
        user: { ...fullUser, role: fullUser.userType?.name || "Player" },
        isAuthenticated: true,
      });
    } catch (error) {
      console.error("Error obteniendo perfil:", error);
      await get().logout();
    }
  },

  initAuth: async () => {
    try {
      const storedToken = await AsyncStorage.getItem("pingpong_token");
      if (storedToken) {
        set({ token: storedToken, isAuthenticated: true });
        await get().fetchProfile();
      }
    } catch (e) {
      console.error("Error inicializando auth", e);
    } finally {
      set({ isLoading: false });
    }
  },
}));
